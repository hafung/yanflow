cmake_minimum_required(VERSION 3.16)
project(yanflow-csc-worker LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "CSC worker requires Windows x64 / MSVC")
endif()
if(NOT EXISTS "${ONNXRUNTIME_DIR}/include/onnxruntime_cxx_api.h")
    message(FATAL_ERROR "ONNXRUNTIME_DIR must point to the verified ONNX Runtime archive")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(yanflow-csc-worker main.cpp ../csc-protocol.h ../text-correction.h ../text-correction.cpp)
target_compile_features(yanflow-csc-worker PRIVATE cxx_std_17)
target_compile_options(yanflow-csc-worker PRIVATE /W4 /O2 /utf-8 /EHsc)
target_include_directories(yanflow-csc-worker PRIVATE "${ONNXRUNTIME_DIR}/include")
